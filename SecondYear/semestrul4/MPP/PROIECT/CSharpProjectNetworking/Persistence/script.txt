CREATE TABLE IF NOT EXISTS participanti (
                    part_id INTEGER PRIMARY KEY AUTOINCREMENT,
                    name TEXT NOT NULL,
                    cnp text NOT NULL
                );
                
create table if not exists organizatori
(
    org_id   integer not null
        primary key,
    username varchar(100),
    password varchar(100)
);

create table if not exists categorievarsta
(
    cv_id     integer not null
        primary key,
    varstamin integer,
    varstamax integer
);

create table if not exists probe
(
    proba_id integer not null
        primary key,
    tip      varchar(100),
    varsta   integer
        references categorievarsta(cv_id)
);

create table if not exists inscrieri
(
    participant integer
        references participanti(part_id),
    proba       integer
        references probe(proba_id)
);
